let spinning=false, interval=null, timeout=null;


function tick(){
const [p,s1,s2] = uniqueSkills(3);
els.race.textContent = rand(RACES);
els.primary.textContent = p;
els.secondary1.textContent = s1;
els.secondary2.textContent = s2;
els.quest.textContent = rand(QUESTS);
}


function spin(){
if(spinning) return;
spinning = true;
els.spinBtn.disabled = true;
if(navigator.vibrate) navigator.vibrate(10);
tick();
interval = setInterval(tick, 70);
timeout = setTimeout(()=>{
clearInterval(interval);
tick();
spinning=false;
els.spinBtn.disabled = false;
if(navigator.vibrate) navigator.vibrate([8,40,8]);
}, 2000);
}


function clearAll(){
['race','primary','secondary1','secondary2','quest'].forEach(k=>els[k].textContent='—');
}


async function copyResult(){
const text = `Race: ${els.race.textContent}\nPrimary Skill: ${els.primary.textContent}\nSecondary Skill 1: ${els.secondary1.textContent}\nSecondary Skill 2: ${els.secondary2.textContent}\nQuest: ${els.quest.textContent}`;
try{
await navigator.clipboard.writeText(text);
alert('Copied loadout to clipboard!');
}catch(e){
alert('Could not copy to clipboard in this environment.');
}
}


els.spinBtn.addEventListener('click', spin);
els.clearBtn.addEventListener('click', clearAll);
els.copyBtn.addEventListener('click', copyResult);


// Register service worker for offline use
if('serviceWorker' in navigator){
window.addEventListener('load', ()=>{
navigator.serviceWorker.register('./sw.js').catch(()=>{});
});
}
